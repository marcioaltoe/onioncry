"""Run with: rtk proxy python3 <this-file> Q2|Q3|Q4|Q5 from the Run root."""
import hashlib
import json
from pathlib import Path
import shlex
import subprocess
import sys
import tempfile
import traceback

ROOT = Path.cwd()
EVIDENCE = Path(__file__).resolve().parent
REPORT = EVIDENCE.parent.parent / 'qa-report-2026-11-06.md'
BINARY = ROOT / 'target/debug/onioncry'
RULE = 'cleanarch/artifact-placement'
ROW = sys.argv[1]
LOG = []


def run(args, cwd, expected=0):
    command = ['rtk', 'proxy', str(BINARY), *args]
    result = subprocess.run(command, cwd=cwd, text=True, capture_output=True)
    LOG.append(f'cwd: {cwd}\ncommand: {shlex.join(command)}\nexit: {result.returncode}\nstdout:\n{result.stdout}\nstderr:\n{result.stderr}')
    assert result.returncode == expected, f'{args}: exit {result.returncode}, expected {expected}'
    return result


def config(root, severity='error', project_root='src', context='contexts', overrides=None, architecture=None, rules=None):
    data = {'version': 1, 'project': {'root': project_root, 'include': ['**/*.ts'], 'exclude': []},
            'architecture': architecture or {'mode': 'cleanArchitecture', 'cleanArchitecture': {'contextRoot': context}},
            'rules': rules if rules is not None else {RULE: severity}, 'overrides': overrides or []}
    # JSONC comments exercise the established public configuration contract.
    text = '// Generic QA fixture; no consumer source.\n' + json.dumps(data, indent=2) + '\n'
    (root / '.onioncryrc.jsonc').write_text(text)
    LOG.append('fixture .onioncryrc.jsonc:\n' + text)


def source(root, path, text='export const value = 1;\n'):
    target = root / path
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_text(text)
    LOG.append(f'fixture {path}:\n{text}')


def snapshot(root):
    return {str(p.relative_to(root)): hashlib.sha256(p.read_bytes()).hexdigest()
            for p in sorted(root.rglob('*')) if p.is_file()}


def check(root, expected=0, *extra):
    result = run(['check', '--format', 'json', *extra], root, expected)
    assert not result.stderr, 'JSON diagnostic unexpectedly on stderr'
    return json.loads(result.stdout)


def stable(root, expected=0, *extra):
    before = snapshot(root)
    one = check(root, expected, *extra)
    two = check(root, expected, *extra)
    assert one == two, 'fresh invocation differs'
    assert snapshot(root) == before, 'read-only CLI changed fixture bytes'
    LOG.append('Independent confirmation: fresh process JSON identical; all fixture SHA-256 values unchanged.\n' + json.dumps(before, indent=2))
    return one


def assert_placement(v, path, layer, role, prefix='contexts/tax', severity='warn'):
    boundary = f'{prefix}/{layer}/{role}'
    assert v['file'].endswith('/' + path), (v, path)
    assert v['rule'] == RULE and v['severity'] == severity
    assert v['toLayer'] == layer
    assert v['message'] == f'cleanArchitecture artifact "{role}" should live in the {layer} boundary {boundary}'
    assert v['suggestion'] == f'move this {role} artifact to {boundary} or turn cleanarch/artifact-placement off with an override while migrating'


def positive(root):
    config(root, 'warn')
    cases = [
        ('application/qa-placement-probe.use-case.ts', 'application', 'use-cases'),
        ('application/billing/nested.use-case.ts', 'application', 'use-cases'),
        ('infra/audit.repository.ts', 'infra', 'repositories'),
        ('domain/money.entity.ts', 'domain', 'entities'),
        ('domain/money.value-object.ts', 'domain', 'value-objects'),
        ('application/catalog.port.ts', 'application', 'ports'),
        ('infra/http.adapter.ts', 'infra', 'adapters'),
        ('application/billing.service.ts', 'application', 'services'),
    ]
    for relative, _, _ in cases:
        source(root, 'src/contexts/tax/' + relative)
    # Unlike the seven placement-classified suffix roles above, handlers have
    # no default Clean Architecture artifact folder. Keep their generic finding.
    source(root, 'src/contexts/tax/infra/http.handler.ts')
    result = stable(root, 1, '--fail-on', 'warning')
    assert result['status'] == 'fail'
    assert result['summary']['warningCount'] == len(cases) + 1
    assert result['summary']['errorCount'] == 0
    assert len(result['violations']) == len(cases) + 1
    by_path = {v['file']: v for v in result['violations']}
    for path, layer, role in cases:
        file = 'src/contexts/tax/' + path
        assert_placement(by_path[str((root/file).resolve())], file, layer, role)
    handler = by_path[str((root/'src/contexts/tax/infra/http.handler.ts').resolve())]
    assert handler['message'] == 'cleanArchitecture artifact "infra" should live in the infra boundary contexts/tax/infra/<artifact-folder>'
    assert handler['suggestion'] == 'move this infra artifact to contexts/tax/infra/<artifact-folder> or turn cleanarch/artifact-placement off with an override while migrating'
    default = check(root)
    assert default['status'] == 'pass' and default['violations'] == result['violations']
    explain = json.loads(run(['explain', 'src/contexts/tax/application/qa-placement-probe.use-case.ts', '--format', 'json'], root).stdout)
    assert any(v['rule'] == RULE for v in explain['violations'])
    LOG.append('PASS: all seven placement-classified default suffix roles plus nested use-case report exact boundaries; handler retains generic direct-layer finding; warning/default thresholds and explain agree.')


def negative(root):
    config(root, overrides=[{'files': ['contexts/tax/application/probe.use-case.ts'], 'rules': {RULE: 'off'}}])
    for layer, role, filename in [('application', 'use-cases', 'probe.use-case.ts'), ('infra', 'repositories', 'audit.repository.ts'), ('domain', 'entities', 'money.entity.ts')]:
        source(root, f'src/contexts/tax/{layer}/{role}/{filename}')
        source(root, f'src/contexts/tax/{layer}/{role}/billing/{filename}')
    clean = stable(root, 0, '--fail-on', 'warning')
    assert clean['violations'] == []
    source(root, 'src/contexts/tax/application/probe.use-case.ts')
    source(root, 'src/contexts/tax/infra/audit.repository.ts')
    overridden = stable(root, 1)
    assert len(overridden['violations']) == 1
    assert_placement(overridden['violations'][0], 'src/contexts/tax/infra/audit.repository.ts', 'infra', 'repositories', severity='error')
    # Contextless base and source-prefixed layer-first paths retain diagnostics.
    with tempfile.TemporaryDirectory(prefix='onioncry-qa-base-') as temporary:
        base = Path(temporary).resolve()
        config(base, project_root='.', context='src/contexts')
        source(base, 'src/application/base.use-case.ts')
        source(base, 'src/application/billing/create.use-case.ts')
        result = stable(base, 1)
        assert len(result['violations']) == 2
        assert all(v['rule'] == RULE and 'src/application/use-cases' in v['suggestion'] for v in result['violations'])
    # Explicit project root '.' and nested contextRoot remain supported.
    with tempfile.TemporaryDirectory(prefix='onioncry-qa-root-') as temporary:
        alternate = Path(temporary).resolve()
        config(alternate, project_root='.', context='src/contexts')
        source(alternate, 'src/contexts/tax/application/use-cases/probe.use-case.ts')
        assert stable(alternate, 0)['violations'] == []
    LOG.append('PASS: six direct/nested controls, isolated off override, contextless and layer-first findings, alternate root, repeat invocations and read-only persistence.')


def contracts(root):
    for args in [['--version'], ['--help'], ['check', '--help'], ['explain', '--help']]:
        result = run(args, root)
        assert not result.stderr
    config(root)
    source(root, 'src/contexts/tax/application/probe.use-case.ts')
    before = snapshot(root)
    report = check(root, 1)
    assert len(report['violations']) == 1 and report['violations'][0]['severity'] == 'error'
    assert report['summary']['errorCount'] == 1
    explain = json.loads(run(['explain', 'src/contexts/tax/application/probe.use-case.ts', '--format', 'json'], root).stdout)
    assert explain['violations'] == report['violations']
    sarif_result = run(['check', '--format', 'sarif'], root, 1)
    assert not sarif_result.stderr
    sarif = json.loads(sarif_result.stdout)
    assert sarif['version'] == '2.1.0' and len(sarif['runs']) == 1
    assert sarif['runs'][0]['results'][0]['ruleId'] == RULE
    assert len(sarif['runs'][0]['results']) == 1
    llm = run(['check', '--llm-mode'], root, 1)
    assert not llm.stderr and RULE in llm.stdout
    assert 'onioncry-llm-report v1 revision: dcf0984cba5b' in llm.stdout
    pretty = run(['check', '--tips'], root, 1)
    assert RULE in pretty.stdout
    run(['explain', 'src/contexts/tax/application/probe.use-case.ts', '--tips'], root)
    assert snapshot(root) == before
    bad = run(['check', '--format', 'unsupported'], root, 2)
    assert bad.stderr and not bad.stdout
    with tempfile.TemporaryDirectory(prefix='onioncry-qa-noconfig-') as temporary:
        missing = run(['check', '--format', 'json'], Path(temporary), 2)
        assert missing.stderr and not missing.stdout
    # Unsuffixed placement diagnostics and unclassified-file remain independent.
    with tempfile.TemporaryDirectory(prefix='onioncry-qa-unsuffixed-') as temporary:
        unsuffixed = Path(temporary).resolve()
        config(unsuffixed)
        source(unsuffixed, 'src/contexts/tax/application/plain.ts', 'export class InvoiceUseCase {}\n')
        result = stable(unsuffixed, 1)
        assert len(result['violations']) == 1 and result['violations'][0]['rule'] == RULE
        assert '<artifact-folder>' in result['violations'][0]['suggestion']
        source(unsuffixed, 'src/tools/plain.ts')
        config(unsuffixed, rules={'cleanarch/unclassified-file': 'error'})
        # Layer-classification rules deliberately do not run without layers.
        config_path = unsuffixed / '.onioncryrc.jsonc'
        data = json.loads('\n'.join(config_path.read_text().splitlines()[1:]))
        data['layers'] = {'application': {'patterns': ['contexts/*/application/**'], 'mayImport': ['application']}}
        config_path.write_text(json.dumps(data, indent=2) + '\n')
        LOG.append('Unclassified-file fixture with explicit layers:\n' + config_path.read_text())
        unclassified = stable(unsuffixed, 1)
        assert len(unclassified['violations']) == 1
        assert unclassified['violations'][0]['rule'] == 'cleanarch/unclassified-file'
        assert unclassified['violations'][0]['file'].endswith('/src/tools/plain.ts')
    with tempfile.TemporaryDirectory(prefix='onioncry-qa-vertical-') as temporary:
        vertical = Path(temporary).resolve()
        config(vertical, architecture={'mode': 'verticalSlice', 'verticalSlice': {'sliceRoot': 'features', 'sliceDepth': 2}}, rules={})
        source(vertical, 'src/features/tax/calculate/index.ts')
        source(vertical, 'src/features/tax/calculate/handlers/probe.handler.ts')
        result = stable(vertical, 0)
        assert not result['violations']
    LOG.append('PASS: help, JSON check/explain agreement, SARIF 2.1.0 envelope, LLM v1 footer, pretty output, 0/1/2 exits, stderr-only operational failures, unsuffixed/class-name control, unclassified rule, Vertical Slice mode. Full SARIF schema validation belongs to the Daemon-passed repository suite, not this envelope check.')


def outside(root):
    reference = ROOT / 'docs/references/2026-11-06-artifact-suffix-placement-evidence.md'
    text = reference.read_text()
    assert 'https://www.jsboundaries.dev/docs/classification/elements/' in text
    assert 'Element patterns should match folders' in text
    LOG.append('Outside origin: JS Boundaries element descriptor documentation, https://www.jsboundaries.dev/docs/classification/elements/ . Cited passage checked offline in the committed reference: element patterns match folders; file descriptors classify file kinds. No live network verification claimed. This is an analogous architecture principle, not proof of OnionCry behavior; the following real-binary comparison supplies that proof.')
    config(root)
    misplaced = 'src/contexts/tax/application/probe.use-case.ts'
    placed = 'src/contexts/tax/application/use-cases/probe.use-case.ts'
    source(root, misplaced)
    result = stable(root, 1)
    assert len(result['violations']) == 1
    assert_placement(result['violations'][0], misplaced, 'application', 'use-cases', severity='error')
    before = (root/misplaced).read_bytes()
    (root/placed).parent.mkdir(parents=True, exist_ok=True)
    (root/misplaced).rename(root/placed)
    LOG.append(f'Operator action: move {misplaced} to {placed}, preserving bytes.')
    assert (root/placed).read_bytes() == before
    assert stable(root, 0)['violations'] == []
    LOG.append('PASS: same suffix and source; only folder placement changes the JSON result from one exact placement error to clean, matching the independently published folder/file-kind distinction.')


def main():
    functions = {'Q2': positive, 'Q3': negative, 'Q4': contracts, 'Q5': outside}
    status = 'pass'
    try:
        with tempfile.TemporaryDirectory(prefix=f'onioncry-qa-{ROW.lower()}-') as temporary:
            functions[ROW](Path(temporary).resolve())
    except Exception:
        status = 'fail'
        LOG.append(traceback.format_exc())
    (EVIDENCE/f'{ROW}.txt').write_text('\n\n'.join(LOG) + '\n')
    report = REPORT.read_text()
    assert f'| {ROW} | pending |' in report
    report = report.replace(f'| {ROW} | pending |', f'| {ROW} | {status} |')
    report += f'\n{ROW} execution: {status}; [full command, stream and fixture evidence](evidence/2026-11-06-cli/{ROW}.txt).\n'
    REPORT.write_text(report)
    print(f'{ROW}: {status}; {EVIDENCE/ROW}.txt')
    if status != 'pass':
        print(LOG[-1])
        sys.exit(1)


if __name__ == '__main__':
    main()
