"""Run with python3 <this-file> <01|02|03|04|05> from the audited repository."""
import hashlib
import json
from pathlib import Path
import subprocess
import sys
import tempfile

ROOT = Path.cwd()
OUT = Path(__file__).resolve().parent
BIN = ROOT / 'target/debug/onioncry'
REPORT = OUT.parent.parent / 'qa-report-2026-11-06.md'
Q = 'frontend/feature-system-query-contract'
D = 'frontend/feature-system-dependency-flow'
SYSTEM = 'packages/frontend/src/systems/billing/'
HOOK = SYSTEM + 'hooks/use-billing.ts'
COMP = SYSTEM + 'components/billing-card.tsx'
ROUTE = 'packages/frontend/src/routes/billing.tsx'
KEYS = "export const keys = { list: () => ['billing'] as const };\n"
OPTIONS = "import { queryOptions } from '@tanstack/react-query';\nimport { keys } from './query-keys';\nexport const options = () => queryOptions({ queryKey: keys.list(), queryFn: async () => [] });\n"
READ = "import { useQuery } from '@tanstack/react-query';\nimport { options } from '../lib/query-options';\nexport const result = useQuery(options());\n"
MUTATION = 'mutation hooks should invalidate relevant queries in onSuccess or onSettled'
HKEY = 'query hooks should reuse factories from lib/query-options.ts'
CKEY = 'components should not own query keys'
RKEY = 'routes should not own query keys'
OPTIMISTIC = 'optimistic cache updates should cancel, snapshot or rollback, and invalidate on settlement'
METHODS = ['invalidateQueries', 'refetchQueries', 'cancelQueries', 'resetQueries', 'removeQueries', 'isFetching', 'getQueriesData', 'setQueriesData']
row = sys.argv[1]
observations = []
cases = []

def call(cwd, *args):
    p = subprocess.run([str(BIN), *args], cwd=cwd, capture_output=True, text=True, timeout=45)
    observations.append({'cwd': str(cwd), 'command': [str(BIN), *args], 'exit': p.returncode, 'stdout': p.stdout, 'stderr': p.stderr})
    (OUT / ('row-' + row + '-commands.json.txt')).write_text(json.dumps(observations, indent=2))
    return p

def snapshot(project):
    return {str(f.relative_to(project)): hashlib.sha256(f.read_bytes()).hexdigest() for f in project.rglob('*') if f.is_file()}

def case(name, additions, expected=(), severity='warn'):
    files = {SYSTEM + 'lib/query-keys.ts': KEYS, SYSTEM + 'lib/query-options.ts': OPTIONS}
    files.update(additions)
    config = {'version': 1, 'project': {'root': '.', 'include': ['packages/frontend/src/**/*'], 'exclude': []}, 'rules': {Q: severity, D: 'warn'}, 'overrides': []}
    files['.onioncryrc.jsonc'] = json.dumps(config, indent=2)
    cases.append({'name': name, 'files': files, 'expected': expected})
    (OUT / ('row-' + row + '-fixtures.json.txt')).write_text(json.dumps(cases, indent=2))
    with tempfile.TemporaryDirectory(prefix='onioncry-qa-') as directory:
        project = Path(directory)
        for path, content in files.items():
            target = project / path
            target.parent.mkdir(parents=True, exist_ok=True)
            target.write_text(content)
        before = snapshot(project)
        first = call(project, 'check', '--format', 'json', '--fail-on', 'warning')
        result = json.loads(first.stdout)
        assert first.returncode == (1 if expected else 0), (name, first)
        assert not first.stderr, (name, first.stderr)
        found = [(v['rule'], v['message']) for v in result['violations']]
        assert sorted(found) == sorted(expected), (name, found, expected)
        for violation in result['violations']:
            assert violation['severity'] == ('warn' if violation['rule'] == D else severity)
            assert violation['file'] and violation['suggestion']
        repeated = call(project, 'check', '--format', 'json', '--fail-on', 'warning')
        assert repeated.returncode == first.returncode and repeated.stdout == first.stdout
        explained = next(iter(additions))
        explanation = call(project, 'explain', explained, '--format', 'json')
        assert explanation.returncode == 0 and not explanation.stderr
        explanation_json = json.loads(explanation.stdout)
        for _, message in expected:
            assert message in explanation.stdout, (name, 'explain missing diagnostic', explanation_json)
        assert snapshot(project) == before, (name, 'CLI changed input project')
        print(name, ': pass; diagnostics:', len(found), '; independent check/explain; input hashes unchanged', flush=True)

def mutation(options):
    return "import { useMutation } from '@tanstack/react-query';\nexport const useExport = () => useMutation({ mutationFn: async () => new Blob(), " + options + ' });\n'

def execute():
    if row == '01':
        case('reported read-and-invalidate hook', {HOOK: READ + "export const update = useMutation({ mutationFn: async () => undefined, onSuccess: () => client.invalidateQueries({ queryKey: keys.list() }) });\n"})
        for method in METHODS:
            for area, path in [('hook', HOOK), ('component', COMP), ('route', ROUTE)]:
                source = (READ if area == 'hook' else '') + "const queryKey = keys.list();\nclient." + method + "({ queryKey: keys.list() });\nclient." + method + "({ queryKey });\nclient['" + method + "']({ 'queryKey': queryKey });\n"
                case(method + ' factory/shorthand/quoted in ' + area, {path: source})
        case('legacy explicit onSettled invalidation', {HOOK: mutation("onSettled: () => client.invalidateQueries({ queryKey: keys.list() })")})
        case('declared empty invalidation', {HOOK: mutation('meta: { invalidates: [] }')})
        case('declared nonempty invalidation', {HOOK: mutation("meta: { invalidates: [['billing']] }")})
    elif row == '02':
        for area, path, message in [('hook', HOOK, HKEY), ('component', COMP, CKEY), ('route', ROUTE, RKEY)]:
            for method in METHODS:
                source = (READ if area == 'hook' else '') + 'client.' + method + "({ queryKey: ['billing'] });\n"
                case('inline filter ' + method + ' in ' + area, {path: source}, [(Q, message)])
        case('inline read options', {HOOK: "import { useQuery } from '@tanstack/react-query';\nexport const result = useQuery({ queryKey: ['billing'] });\n"}, [(Q, HKEY)])
        for name, options in [('missing meta', ''), ('unrelated meta', "meta: { toast: 'saved' }")]:
            case(name, {HOOK: mutation(options)}, [(Q, MUTATION)])
        case('mixed mutations', {HOOK: mutation('meta: { invalidates: [] }') + '\nuseMutation({});'}, [(Q, MUTATION)])
        case('optimistic update without rollback', {HOOK: mutation("meta: { invalidates: [] }, onMutate: () => client.setQueryData(['billing'], [])")}, [(Q, OPTIMISTIC)])
        case('production component adapter import', {COMP: "import { billingApi } from '../adapters/billing-api';\nexport const BillingCard = billingApi;", SYSTEM + 'adapters/billing-api.ts': 'export const billingApi = {};'}, [(D, 'components may not import adapters in the billing system through ../adapters/billing-api')])
    elif row == '03':
        case('comment and string are not keys or mutations', {HOOK: "// queryKey: ['billing']; useMutation({})\nconst label = 'queryKey useMutation(';\nconst meta = { invalidates: [] };\nconst queryKey = ['billing'];\n"})
        case('comment meta cannot contract actual mutation', {HOOK: mutation('/* meta: { invalidates: [] } */')}, [(Q, MUTATION)])
        case('unrelated object cannot contract actual mutation', {HOOK: mutation('') + '\nconst extra = { meta: { invalidates: [] } };'}, [(Q, MUTATION)])
        case('quoted member mutation', {HOOK: "query['useMutation']({ 'meta': { 'invalidates': [] } });"})
        case('member mutation lacking meta', {HOOK: 'query.useMutation({});'}, [(Q, MUTATION)])
        case('nonarray invalidates is declaration not proof', {HOOK: mutation('meta: { invalidates: null }')})
        case('nested invalidates does not contract mutation', {HOOK: mutation('meta: { nested: { invalidates: [] } }')}, [(Q, MUTATION)])
        case('nonfilter key remains ownership', {COMP: 'client.other({ queryKey: keys.list() });'}, [(Q, CKEY)])
        case('later argument key remains ownership', {COMP: 'client.invalidateQueries({}, { queryKey: keys.list() });'}, [(Q, CKEY)])
        case('nested filter key remains ownership', {COMP: "client.invalidateQueries({ queryKey: factory({ queryKey: ['billing'] }) });"}, [(Q, CKEY)])
        case('option factories remain outside cross-file analysis', {HOOK: "import { useMutation } from '@tanstack/react-query';\nuseMutation(mutationOptions());"}, [(Q, MUTATION)])
    elif row == '04':
        version = call(ROOT, '--version')
        assert version.returncode == 0 and 'revision: 2c2bc4a7705c' in version.stdout
        for command in ['check', 'explain']:
            p = call(ROOT, command, '--help')
            assert p.returncode == 0 and not p.stderr and '--format' in p.stdout
        with tempfile.TemporaryDirectory(prefix='onioncry-qa-output-') as directory:
            project = Path(directory)
            (project / '.onioncryrc.jsonc').write_text(json.dumps({'version': 1, 'project': {'root': '.', 'include': ['packages/frontend/src/**/*'], 'exclude': []}, 'rules': {Q: 'warn'}, 'overrides': []}))
            path = project / 'packages/frontend/src/systems/billing/hooks/use-export.ts'
            path.parent.mkdir(parents=True)
            path.write_text(mutation(''))
            before = snapshot(project)
            p = call(project, 'check', '--format', 'json')
            assert p.returncode == 0 and not p.stderr
            data = json.loads(p.stdout)
            assert data['summary']['warningCount'] == 1
            assert data['violations'][0]['message'] == MUTATION
            assert data['violations'][0]['suggestion'] == 'add an onSuccess or onSettled handler that calls invalidateQueries'
            p = call(project, 'check', '--format', 'json', '--fail-on', 'warning')
            assert p.returncode == 1 and not p.stderr
            assert json.loads(p.stdout)['violations'] == data['violations']
            p = call(project, 'explain', str(path.relative_to(project)), '--format', 'json')
            assert p.returncode == 0 and not p.stderr and MUTATION in p.stdout
            p = call(project, 'check', '--format', 'sarif', '--fail-on', 'warning')
            assert p.returncode == 1 and not p.stderr
            sarif = json.loads(p.stdout)
            assert sarif['version'] == '2.1.0' and sarif['runs'][0]['results'][0]['ruleId'] == Q
            p = call(project, 'check', '--llm-mode', '--fail-on', 'warning')
            assert p.returncode == 1 and not p.stderr
            assert 'onioncry-llm-report v1 revision:' in p.stdout and Q in p.stdout
            p = call(project, 'check', '--tips', '--fail-on', 'warning')
            assert p.returncode == 1 and not p.stderr and MUTATION in p.stdout
            assert snapshot(project) == before
            p = call(project, 'check', '--config', 'missing.jsonc', '--format', 'json')
            assert p.returncode == 2 and not p.stdout and p.stderr
            p = call(project, 'check', '--fail-on', 'invalid')
            assert p.returncode == 2 and not p.stdout and p.stderr
            path.write_text('const = ;')
            p = call(project, 'check', '--format', 'json')
            assert p.returncode == 2 and not p.stdout and p.stderr
        print('help, check/explain, JSON, SARIF, LLM footer, tips, 0/1/2 exits and stream separation: pass', flush=True)
    elif row == '05':
        reference = ROOT / 'docs/references/2026-11-06-query-contract-filters-and-mutation-meta.md'
        text = reference.read_text()
        assert 'https://tkdodo.eu/blog/automatic-query-invalidation-after-mutations' in text
        assert 'https://github.com/TkDodo/blog-comments/discussions/105' in text
        assert '{ invalidates: [] }' in text
        for name, options, expected in [('empty invalidates', 'meta: { invalidates: [] }', []), ('missing meta', '', [(Q, MUTATION)]), ('meta without invalidates', "meta: { toast: 'saved' }", [(Q, MUTATION)])]:
            case('published guidance: ' + name, {HOOK: mutation(options)}, expected)
        print('Outside origin: committed published maintainer passage; no network required. Empty declaration accepts; both controls fail.', flush=True)
    else:
        raise ValueError(row)

try:
    execute()
except Exception:
    (OUT / ('row-' + row + '-commands.json.txt')).write_text(json.dumps(observations, indent=2))
    raise
else:
    text = REPORT.read_text().replace('| ' + row + ' | pending |', '| ' + row + ' | pass |', 1)
    detail = 'Actor: frontend developer / CLI operator. Inputs: repository_path = Run Worktree root. Entry point: target/debug/onioncry. All commands and outputs: [command evidence](evidence/2026-11-06-query-contract/row-' + row + '-commands.json.txt).'
    if cases:
        detail += ' Exact synthetic input files and expectations: [fixtures](evidence/2026-11-06-query-contract/row-' + row + '-fixtures.json.txt). ' + str(len(cases)) + ' independent fixture cases passed; fresh check and explain confirmed each outcome; input hashes remained unchanged.'
    text = text.replace('<!-- observations-' + row + ' -->', detail)
    REPORT.write_text(text)
