ROOT = Path.cwd()

events = []

def git(*args, allowed=(0,)):
    result = subprocess.run(['rtk', 'proxy', 'git', '-c', 'core.fsmonitor=false', *args], cwd=ROOT, text=True, capture_output=True, timeout=30)
    events.append({'command': ['git', '-c', 'core.fsmonitor=false', *args], 'exit': result.returncode, 'stdout': result.stdout, 'stderr': result.stderr})
    (OUT / 'audit-commands.json.txt').write_text(json.dumps(events, indent=2))
    assert result.returncode in allowed, events[-1]
    return result.stdout

def complete(row, detail):
    text = REPORT.read_text().replace('| ' + row + ' | pending |', '| ' + row + ' | pass |', 1)
    detail_text = 'Actor: maintainer / documentation reader. Inputs: repository_path = Run Worktree root; commit_range = ' + BASE + '..' + HEAD + '. Surface: repository/docs. Evidence: [audit commands and exact outputs](evidence/2026-11-06-query-contract/audit-commands.json.txt), [audit conclusions](evidence/2026-11-06-query-contract/audit-summary.txt). ' + detail + '\n'
    text = text.replace('<!-- observations-' + row + ' -->', detail_text)
    REPORT.write_text(text)
