All projects are generic temporary fixtures; the analyzed TypeScript is never executed.

def require(condition, message):
    if not condition:
        problems.append(message)

def config(family, setting='error', overrides=None):
    return {'version': 1, 'project': {'root': '.', 'include': ['**/*.ts', '**/*.tsx'], 'exclude': []},
            'rules': {RULE + family: setting}, 'overrides': overrides or []}

def override(family, file, setting):
    return {'files': [file], 'rules': {RULE + family: setting}}

def write(root, file, source):
    p = root / file
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(source)

def fixture(root, cfg, files):
    write(root, '.onioncryrc.jsonc', '// QA generic source fixture\n' + json.dumps(cfg))
    for file, source in files.items():
        write(root, file, source)

def digest(root):
    return {str(p.relative_to(root)): hashlib.sha256(p.read_bytes()).hexdigest()
            for p in sorted(root.rglob('*')) if p.is_file()}

def command(root, args, name, expected_exit):
    before = digest(root)
    try:
        result = subprocess.run(['rtk', 'proxy', str(BINARY)] + args, cwd=root,
                                capture_output=True, text=True, timeout=20)
        code, stdout, stderr = result.returncode, result.stdout, result.stderr
    except subprocess.TimeoutExpired:
        code, stdout, stderr = -999, '', 'subprocess exceeded 20 seconds'
    after = digest(root)
    record = {'case': name, 'command': ['<RUN_BINARY>'] + args, 'exit': code,
              'expected_exit': expected_exit, 'stdout': stdout, 'stderr': stderr,
              'fixture': {str(p.relative_to(root)): p.read_text() for p in sorted(root.rglob('*')) if p.is_file()},
              'sha256_before': before, 'sha256_after': after}
    records.append(record)
    require(code == expected_exit, name + ': exit ' + str(code) + ', expected ' + str(expected_exit))
    require(before == after, name + ': command changed fixture files')
    return record

def relative(file, root):
    p = Path(file)
    return p.relative_to(root).as_posix() if p.is_absolute() and p.is_relative_to(root) else p.as_posix()

def check(root, family, expected, name, strict=True):
    errors = sum(sev == 'error' for _, sev in expected)
    warnings = len(expected) - errors
    fails = errors > 0 or (strict and warnings > 0)
    args = ['check', '--format', 'json'] + (['--fail-on', 'warning'] if strict else [])
    record = command(root, args, name, 1 if fails else 0)
    require(record['stderr'] == '', name + ': unexpected stderr')
    try:
        report = json.loads(record['stdout'])
    except json.JSONDecodeError:
        require(False, name + ': stdout is not JSON')
        return {}
    require(report.get('status') == ('fail' if fails else 'pass'), name + ': status mismatch')
    summary = report.get('summary', {})
    require((summary.get('errorCount'), summary.get('warningCount'), summary.get('violationCount')) ==
            (errors, warnings, len(expected)), name + ': summary mismatch')
    actual = sorted((relative(v['file'], root), v['severity']) for v in report.get('violations', []))
    require(actual == sorted(expected), name + ': diagnostic owner/severity mismatch: ' + repr(actual))
    require(all(v['rule'] == RULE + family for v in report.get('violations', [])), name + ': rule mismatch')
    return report

def explain(root, file, name, expected_count):
    record = command(root, ['explain', file, '--format', 'json'], name, 0)
    require(record['stderr'] == '', name + ': explain stderr')
    data = json.loads(record['stdout'])
    require(len(data.get('violations', [])) == expected_count, name + ': explain violation count')
    return data

def case_file(files, family, name):
    if family == 'layout':
        file, source = ROOT + '/components/' + name + '/card.tsx', 'export const Card = () => null;'
    elif family == 'public-api':
        files[SYSTEM + '/lib/internal.ts'] = 'export const internal = 1;'
        file, source = ROOT + '/routes/' + name + '.ts', "import { internal } from '../systems/billing/lib/internal'; export const route = internal;"
    elif family == 'dependency-flow':
        files[ADAPTER] = 'export const billingApi = {};'
        file, source = SYSTEM + '/components/' + name + '.tsx', "import { billingApi } from '../adapters/billing-api'; export const Card = billingApi;"
    elif family == 'adapter-contract':
        file = ROOT + '/systems/' + name + '/adapters/' + name + '-api.ts'
        source = 'export const ' + name + 'Api = {};'
    else:
        file = SYSTEM + '/hooks/' + name + '.ts'
        source = "import { useMutation } from '@tanstack/react-query'; export const mutate = () => useMutation({ mutationFn: async () => undefined });"
    files[file] = source
    return file

        for args in [['--version'], ['--help'], ['check', '--help'], ['explain', '--help']]:
            r = command(root, args, 'identity/help ' + ' '.join(args), 0)
            require(r['stderr'] == '', 'help/version stderr')
        fixture(root, config('adapter-contract', 'warn'), {ADAPTER: 'export const billingApi = {}; export class BillingApiError {}'})
        report = check(root, 'adapter-contract', [(ADAPTER, 'warn')], 'warning default', False)
        check(root, 'adapter-contract', [(ADAPTER, 'warn')], 'warning strict', True)
        data = explain(root, ADAPTER, 'explain same warning', 1)
        require(data['violations'][0]['rule'] == report['violations'][0]['rule'], 'explain rule disagrees')
        require(data['violations'][0]['message'] == report['violations'][0]['message'], 'explain message disagrees')
        for args, name in [(['check'], 'pretty'), (['check', '--format', 'sarif'], 'sarif'), (['check', '--llm-mode'], 'llm')]:
            r = command(root, args, name, 0)
            require(not r['stderr'], name + ': stderr')
            if name == 'sarif':
                doc = json.loads(r['stdout'])
                require(doc['version'] == '2.1.0' and len(doc['runs'][0]['results']) == 1, 'SARIF envelope/count')
            elif name == 'llm':
                require('onioncry-llm-report v1 revision:' in r['stdout'].splitlines()[-1], 'LLM completeness footer')
            else:
                require('BillingApiError' in r['stdout'], 'pretty diagnostic absent')
        bad = command(root, ['check', '--fail-on', 'invalid'], 'invalid threshold', 2)
        require(not bad['stdout'] and 'error:' in bad['stderr'], 'bad usage stream separation')
        write(root, '.onioncryrc.jsonc', '{ invalid jsonc')
        bad = command(root, ['check', '--format', 'json'], 'invalid config', 2)
        require(not bad['stdout'] and bool(bad['stderr']), 'config error stream separation')
        write(root, '.onioncryrc.jsonc', json.dumps(config('adapter-contract')))
        write(root, ADAPTER, 'export const billingApi = {}; export class BillingApiError extends Error {}')
        check(root, 'adapter-contract', [], 'recover clean strict')
        check(root, 'adapter-contract', [], 'fresh process clean replay')
        explain(root, ADAPTER, 'explain clean recovery', 0)

def run_row(rid):
    global records, problems
    records, problems = [], []
    try:
        globals()[rid]()
    except Exception as error:
        problems.append(type(error).__name__ + ': ' + str(error))
    result = {'row': rid, 'binary': str(BINARY), 'observations': records, 'problems': problems,
              'command_count': sum('command' in r for r in records), 'status': 'fail' if problems else 'pass'}
    (OUT / (rid + '.json')).write_text(json.dumps(result, indent=2) + '\n')
    text = REPORT.read_text()
    text = text.replace('| ' + rid + ' | pending |', '| ' + rid + ' | ' + result['status'] + ' |')
    start = text.index('### ' + rid + '\n')
    end = text.find('\n### ', start + 1)
    if end < 0: end = text.index('\n## Findings', start)
    detail = DESCRIPTIONS[rid] + ' Observed ' + str(result['command_count']) + ' fresh CLI subprocesses. '
    detail += 'Evidence: [' + rid + '.json](evidence/' + rid + '.json), with exact arguments, complete stdout/stderr, exits, fixture source/config and before/after SHA-256 maps. '
    detail += 'Independent confirmation: fresh explain or threshold replay; fixture hashes show read-only persistence. Reproduction: `rtk proxy python3 evidence/cli-probes.py.txt <RUN_BINARY> ' + rid + '` from qa/. '
    detail += ('All assertions passed.' if not problems else 'Failures: ' + '; '.join(problems))
    old = 'Steps, expected observable, independent confirmation and evidence: pending execution. Each public CLI subprocess starts fresh; checks are read-only and fixture digests confirm persistence.'
    text = text[:start] + text[start:end].replace(old, detail) + text[end:]
